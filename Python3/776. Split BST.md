## Step 1. Initial Solution

- まず問題文を理解するのに10分もかけてしまった
- target_value≤のBSTと>のBSTに分ける問題
    - 構造はそのままにしてノードの左右の枝をそのまま切って返せば良い
- 大まかには
    - target_valの大小で分ける
    - それぞれのtreeの大きさを比較する
    - 同じなら右側を返す
    
    という流れ
    
- 大きさ判定を分離して実装したが上手いこと合わせてやるのもありかもしれない
- Noneの判定をどこでするか悩んだが一番シンプルな見た目の実装にした
- rootを壊さないようにdeepcopyするところだけ少しミスしたが後は問題なく実装できた

```python
from lintcode import (
    TreeNode,
)
import copy

"""
Definition of TreeNode:
class TreeNode:
    def __init__(self, val):
        self.val = val
        self.left, self.right = None, None
"""

class Solution:
    """
    @param root: the given tree
    @param v: the target value
    @return: the root TreeNode after splitting
    """
    def split_b_s_t(self, root: TreeNode, v: int) -> TreeNode:
        def split_bst_helper(root: TreeNode, v: int) -> (TreeNode, TreeNode):
            if root is None:
                return None, None
            if root.val <= v:
                split_left_tree, split_right_tree = split_bst_helper(root.right, v)
                root.right = split_left_tree
                return root, split_right_tree
            else:
                split_left_tree, split_right_tree = split_bst_helper(root.left, v)
                root.left = split_right_tree
                return split_left_tree, root
        
        def bst_size(root: TreeNode) -> int:
            nodes = [root]
            count = 0
            while nodes:
                node = nodes.pop()
                if node is None:
                    continue
                nodes.append(node.left)
                nodes.append(node.right)
                count += 1
            return count

        left_tree, right_tree = split_bst_helper(copy.deepcopy(root), v)
        left_tree_size, right_tree_size = bst_size(left_tree), bst_size(right_tree)
        if left_tree_size > right_tree_size:
            return left_tree
        return right_tree
```

### Complexity Analysis

- 時間計算量：O(N)
- 空間計算量：O(N)

```python
from lintcode import (
    TreeNode,
)
import copy

class Solution:
    def split_b_s_t(self, root: TreeNode, v: int) -> TreeNode:
        def split_bst_helper(root: TreeNode) -> tuple[TreeNode, TreeNode]:
            if root is None:
                return None, None
            if root.val <= v:
                split_below_v, split_above_v = split_bst_helper(root.right)
                root.right = split_below_v
                return root, split_above_v
            else:
                split_below_v, split_above_v = split_bst_helper(root.left)
                root.left = split_above_v
                return split_below_v, root

        def tree_sizeof(root: TreeNode) -> int:
            nodes = [root]
            count = 0
            while nodes:
                node = nodes.pop()
                if node is None:
                    continue
                nodes.append(node.left)
                nodes.append(node.right)
                count += 1
            return count

        left_root, right_root = split_bst_helper(copy.deepcopy(root))
        if tree_sizeof(left_root) > tree_sizeof(right_root):
            return left_root
        return right_root
```
