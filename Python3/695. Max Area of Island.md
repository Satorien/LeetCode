## Step 1. Initial Solution

- Number of Islandsの時と同じように走査して行って、BFSをするときに島の大きさを調べれば良い
    - 10分程度でほとんど完成したが、gridの中身がstringじゃなくintであることに気づくまで10分以上かかってしまった

```python
class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        visited: Set[Tuple[int]] = set()
        max_area: int = 0

        def unvisited_land(i: int, j: int) -> bool:
            return grid[i][j] == 1 and (i, j) not in visited
        
        def check_island_from(start_i: int, start_j: int) -> int:
            to_visit: List[Tuple[int]] = [(start_i, start_j)]
            area: int = 0
            while to_visit:
                i, j = to_visit.pop()
                if not (0 <= i < len(grid) and 0 <= j < len(grid[i])):
                    continue
                if unvisited_land(i, j):
                    area += 1
                    visited.add((i, j))
                    to_visit.append((i + 1, j))
                    to_visit.append((i - 1, j))
                    to_visit.append((i, j - 1))
                    to_visit.append((i, j + 1))
            return area

        for i in range(len(grid)):
            for j in range(len(grid[i])):
                if not unvisited_land(i, j):
                    continue
                area = check_island_from(i, j)
                if area > max_area:
                    max_area = area
        return max_area
```

### Complexity Analysis

- 時間計算量：O(mn)
- 空間計算量：O(mn)

## Step 2. Alternatives

- 他の方法だとNumber of Islandsと同じようにDFS（Stackか再帰）やUnion Findが選択肢として思いついていたい
    - Union Findは知らないので調べる
    - やや複雑になってしまったが理解は出来た

```python
class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        land_to_parent = {}
        root_to_area = {}
        max_area = 0

        for i in range(len(grid)):
            for j in range(len(grid[i])):
                if grid[i][j] == 1:
                    land_to_parent[(i, j)] = (i, j)
                    root_to_area[(i, j)] = 1
        
        def find_root(x: Tuple[int]) -> Tuple[int]:
            if x == land_to_parent[x]:
                return x
            land_to_parent[x] = find_root(land_to_parent[x])
            return land_to_parent[x]

        def unite_and_get_area(x: Tuple[int], y: Tuple[int]) -> int:
            root_x = find_root(x)
            root_y = find_root(y)
            if root_x == root_y:
                return 0
            if root_to_area[root_x] >= root_to_area[root_y]:
                land_to_parent[root_y] = root_x
                root_to_area[root_x] += root_to_area[root_y]
                return root_to_area[root_x]
            else:
                land_to_parent[root_x] = root_y
                root_to_area[root_y] += root_to_area[root_x]
                return root_to_area[root_y]

        for i in range(len(grid)):
            for j in range(len(grid[i])):
                if grid[i][j] == 1:
                    if not max_area:
                        max_area = 1
                    for move_i, move_j in [[0, 1], [1, 0], [0, -1], [-1, 0]]:
                        next_i, next_j = i + move_i, j + move_j
                        if not (0 <= next_i < len(grid) and \
                            0 <= next_j < len(grid[next_i]) and \
                            grid[next_i][next_j] == 1):
                            continue
                        area = unite_and_get_area((i,j), (next_i, next_j))
                        if area > max_area:
                            max_area = area
        return max_area
```

## Step 3. Final Solution

- 変数・関数名を見直して、to_checkに入れる前に判定を行うように修正

```python
class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        visited_lands: Set[Tuple[int]] = set()
        rows: int = len(grid)
        columns: int = len(grid[0])
        directions: List[List[int]] = [[1, 0], [0, 1], [-1, 0], [0, -1]]

        def unvisited_land(i: int, j: int) -> bool:
            return grid[i][j] and (i, j) not in visited_lands
        
        def inbound(i: int, j: int) -> bool:
            return 0 <= i < rows and 0 <= j < columns

        def traverse_and_get_area(start: Tuple[int]) -> int:
            visited_lands.add(start)
            area = 0
            to_check: List[Tuple[int]] = [start]
            while to_check:
                i, j = to_check.pop()
                area += 1
                for delta_i, delta_j in directions:
                    next_i, next_j = i + delta_i, j + delta_j
                    if inbound(next_i, next_j) and unvisited_land(next_i, next_j):
                        visited_lands.add((next_i, next_j))
                        to_check.append((next_i, next_j))
            return area

        max_area: int = 0
        for i in range(rows):
            for j in range(columns):
                if not unvisited_land(i, j):
                    continue
                area = traverse_and_get_area((i, j))
                max_area = max(max_area, area)
        return max_area
```
